function renderMermaidDiagram(mermaidCode, outputBox) { outputBox.innerHTML = ""; // SANITIZE: Fix common Mermaid syntax errors from AI mermaidCode = mermaidCode.replace(/([A-Za-z0-9]+)\s+([A-Za-z0-9]+)\s*\[/g, '$1_$2['); mermaidCode = mermaidCode.replace(/–>/g, '-->'); mermaidCode = mermaidCode.replace(/—>/g, '-->'); mermaidCode = mermaidCode.replace(/=>/g, '-->'); const wrapper = document.createElement('div'); wrapper.style.cssText = 'display:flex; flex-direction:column; gap:1rem; width:100%;'; // Extract mermaid code block and text let codeOnly = mermaidCode; let explanation = ''; let notes = ''; let keyPoints = ''; // Try to extract code between ```mermaid and ``` const mermaidBlockMatch = mermaidCode.match(/```mermaid\s*([\s\S]*?)```/); if (mermaidBlockMatch) { codeOnly = mermaidBlockMatch[1].trim(); const beforeCode = mermaidCode.split('```mermaid')[0].trim(); if (beforeCode) explanation = beforeCode; const afterCodeParts = mermaidCode.split('```'); if (afterCodeParts.length > 2) { notes = afterCodeParts[2].trim(); } } else { // Try without mermaid label const codeBlockMatch = mermaidCode.match(/```\s*([\s\S]*?)```/); if (codeBlockMatch) { codeOnly = codeBlockMatch[1].trim(); const beforeCode = mermaidCode.split('```')[0].trim(); if (beforeCode) explanation = beforeCode; } } // Extract Key Points from notes if (notes) { const keyPointsMatch = notes.match(/Key\s*Points[^:]*:([\s\S]*)/i); if (keyPointsMatch) { keyPoints = keyPointsMatch[1].trim(); notes = notes.replace(/Key\s*Points[^:]*:([\s\S]*)/i, '').trim(); } } // Validate mermaid code const validMermaidKeywords = ['graph', 'flowchart', 'mindmap', 'timeline', 'sequenceDiagram', 'classDiagram', 'stateDiagram', 'erDiagram', 'pie', 'gantt', 'journey', 'gitGraph']; const isValidMermaid = validMermaidKeywords.some(keyword => codeOnly.toLowerCase().startsWith(keyword)); // Render explanation if present if (explanation) { const explanationDiv = document.createElement('div'); explanationDiv.className = 'diagram-explanation'; explanationDiv.innerHTML = `

📖 About this Diagram:

${explanation}

`; wrapper.appendChild(explanationDiv); } // Create mermaid div if valid if (isValidMermaid && codeOnly.trim()) { const container = document.createElement('div'); container.className = 'mermaid-container'; container.style.cssText = 'background: linear-gradient(135deg, rgba(255,255,255,0.95), rgba(248,250,252,0.95)); padding:1.5rem;'; const mermaidDiv = document.createElement('div'); mermaidDiv.className = 'mermaid'; mermaidDiv.textContent = codeOnly; mermaidDiv.style.cssText = 'background:#fff; padding:1rem; border-radius:12px; box-shadow:0 2px 8px rgba(0,0,0,0.1);'; container.appendChild(mermaidDiv); wrapper.appendChild(container); outputBox.appendChild(wrapper); // ✅ FIX: Use mermaid.render() with unique ID instead of mermaid.run() const diagramId = 'mermaid-diagram-' + Date.now(); mermaid.render(diagramId, codeOnly) .then(result => { // result.svg contains the rendered SVG string mermaidDiv.innerHTML = result.svg; // Remove the mermaid class to prevent re-rendering mermaidDiv.classList.remove('mermaid'); }) .catch(err => { console.error('Mermaid rendering error:', err); // Show error message and fallback const errorDiv = document.createElement('div'); errorDiv.style.cssText = 'padding:1rem; background:rgba(239,68,68,0.1); border:1px solid rgba(239,68,68,0.3); border-radius:12px; color:#dc2626; margin:1rem 0;'; errorDiv.innerHTML = '⚠ Diagram could not be rendered.
Error: ' + escapeHtml(err.message || 'Unknown error') + '

Showing text-based fallback below.'; wrapper.insertBefore(errorDiv, container); // Fallback to enhanced SVG renderEnhancedSVGFallback(codeOnly, outputBox, explanation, notes, keyPoints); }); } else { // If not valid Mermaid, render enhanced SVG fallback renderEnhancedSVGFallback(codeOnly, outputBox, explanation, notes, keyPoints); return; } // Render key points if present if (keyPoints) { const keyPointsDiv = document.createElement('div'); keyPointsDiv.className = 'diagram-notes'; keyPointsDiv.style.cssText = 'background: linear-gradient(135deg, rgba(245,158,11,0.08), rgba(245,158,11,0.03)); border-left: 3px solid #F59E0B;'; const pointsList = keyPoints.split('\n').filter(p => p.trim()).map(p => { const clean = p.replace(/^[-•*]\s*/, '').trim(); return clean ? `
  • ${clean}
    • ` : ''; }).join(''); keyPointsDiv.innerHTML = `

    🎯 Key Points to Remember:

      ${pointsList}
    `; wrapper.appendChild(keyPointsDiv); } // Render additional notes if present if (notes) { const notesDiv = document.createElement('div'); notesDiv.className = 'diagram-notes'; notesDiv.innerHTML = `

    📝 Additional Notes:

    ${notes}

    `; wrapper.appendChild(notesDiv); } }
